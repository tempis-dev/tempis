Reactive Data

Dynamically update items, categories, and bands via props.

How It Works

The items, categories, and bands props are synced to the core instance via useEffect. When a prop changes, the component calls the corresponding setter (setItems, setCategories, setBands) on the existing instance — no destroy/recreate needed.

Dynamic Add / Remove

import { useState } from 'react';
import { TempisTimeline } from '@tempis/react';

function App() {
  const [items, setItems] = useState([
    { id: 1, label: 'Task A', start: '2026-02-01', end: '2026-02-10' },
  ]);

  let nextId = items.length + 1;

  function addItem() {
    setItems(prev => [
      ...prev,
      {
        id: nextId++,
        label: `Task ${nextId}`,
        start: '2026-02-12',
        end: '2026-02-20'
      }
    ]);
  }

  function removeLastItem() {
    setItems(prev => prev.slice(0, -1));
  }

  return (
    <div>
      <button onClick={addItem}>Add Item</button>
      <button onClick={removeLastItem}>Remove Last</button>
      <TempisTimeline
        items={items}
        options={{
          responsive: true,
          range: { start: '2026-01-25', end: '2026-03-01', position: 'bottom' }
        }}
        height={300}
      />
    </div>
  );
}

Toggle Categories

import { useState } from 'react';
import { TempisTimeline } from '@tempis/react';

function App() {
  const allCategories = [
    { name: 'dev',    label: 'Development', style: { backgroundColor: '#6366f1' } },
    { name: 'design', label: 'Design',      style: { backgroundColor: '#f43f5e' } },
  ];

  const [categories, setCategories] = useState(allCategories);

  function toggleDesign() {
    setCategories(prev =>
      prev.some(c => c.name === 'design')
        ? prev.filter(c => c.name !== 'design')
        : [...prev, allCategories[1]]
    );
  }

  return (
    <div>
      <button onClick={toggleDesign}>Toggle Design</button>
      <TempisTimeline
        items={[
          { id: 1, label: 'Auth',       start: '2026-02-01', end: '2026-02-14', category: 'dev' },
          { id: 2, label: 'Wireframes', start: '2026-02-05', end: '2026-02-12', category: 'design' },
        ]}
        categories={categories}
        options={{
          responsive: true,
          legend: { position: 'top' },
          range: { start: '2026-01-25', end: '2026-03-01', position: 'bottom' }
        }}
        height={300}
      />
    </div>
  );
}
